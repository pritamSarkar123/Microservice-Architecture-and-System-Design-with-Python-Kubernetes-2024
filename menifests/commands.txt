cd menifests

// the sequence is very important
// also check k9s and wait until one service starts running
kubectl apply -f ./redis/

kubectl apply -f ./rabbitmq/
kubectl apply -f ./mongo/
kubectl apply -f ./auth/

kubectl apply -f ./notifier/
kubectl apply -f ./converter/




/// drop everything
kubectl delete deployment --all 
kubectl delete service --all 
kubectl delete secret --all 
kubectl delete configmap --all 
kubectl delete ingress --all 
kubectl delete pvc --all 
kubectl delete statefulset --all






// enable required minikube addons
minikube addons list 
minikube addons enable ingress
minikube addons enable ingress-dns
minikube tunnel // closing it will stop the tunnel


/// change number of replicas 
// here for example taking gateway service  gateway-xyzv.. is the pod name, check pod name in the k9s
